cmake_minimum_required(VERSION 3.20)
project(smartbox_mirror_capture C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT EXISTS "${UXPLAY_SOURCE}/lib/raop.h")
  message(FATAL_ERROR "Set UXPLAY_SOURCE to the pinned UxPlay checkout")
endif()
# Build the protocol libraries only: no renderer, GStreamer, or upstream UI.
foreach(component dns_sd llhttp playfair)
  add_subdirectory("${UXPLAY_SOURCE}/lib/${component}" "${component}")
endforeach()
add_subdirectory("${UXPLAY_SOURCE}/lib" airplay)
add_executable(mirror-capture receiver.c)
target_include_directories(mirror-capture PRIVATE "${UXPLAY_SOURCE}/lib")
target_compile_options(mirror-capture PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(mirror-capture PRIVATE airplay)
